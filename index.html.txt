  <!-- FIREBASE INTEGRATION ENGINE (MANDATORY STORAGE ARCHITECTURE) -->
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
    import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
    import { getFirestore, doc, setDoc, getDoc, collection } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

    let db = null;
    let auth = null;
    let currentUser = null;
    const appId = typeof __app_id !== 'undefined' ? __app_id : 'lets-help-roy';

    async function initFirebaseStorage() {
      const dbStatus = document.getElementById('db-status-badge');
      
      if (typeof __firebase_config === 'undefined' || !__firebase_config) {
        if (dbStatus) dbStatus.innerHTML = `<i class="fa-solid fa-circle-info text-amber-500"></i> Playing offline`;
        return;
      }

      try {
        const firebaseConfig = JSON.parse(__firebase_config);
        const app = initializeApp(firebaseConfig);
        auth = getAuth(app);
        db = getFirestore(app);

        // Authenticate (RULE 3)
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }

        onAuthStateChanged(auth, async (user) => {
          if (user) {
            currentUser = user;
            if (dbStatus) dbStatus.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-500 text-base"></i> Online Save Ready`;
            
            // Fetch potential previous progress from Firestore
            await loadProgressFromCloud();
          }
        });
      } catch (err) {
        console.error("Firebase init failed:", err);
        if (dbStatus) dbStatus.innerHTML = `<i class="fa-solid fa-triangle-exclamation text-red-500"></i> Offline mode`;
      }
    }

    // Load progress from private path (RULE 1)
    async function loadProgressFromCloud() {
      if (!db || !currentUser) return;
      try {
        const docRef = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'settings', 'profile');
        const docSnap = await getDoc(docRef);
        if (docSnap.exists()) {
          const data = docSnap.data();
          if (data.stationsCompleted) {
            window.stationsCompleted = data.stationsCompleted;
          }
          if (typeof data.totalStars !== 'undefined') {
            window.totalStars = data.totalStars;
            window.updateHUDAwards();
          }
          if (data.cadetName) {
            const inputField = document.getElementById('cadet-name-input');
            if (inputField) {
              inputField.value = data.cadetName;
              window.syncBadgeName();
            }
          }
          window.updateRouteMap();
        }
      } catch (e) {
        console.error("Failed loading online progress", e);
      }
    }

    // Save progress to private path (RULE 1)
    async function saveProgressToCloud() {
      if (!db || !currentUser) return;
      try {
        const cadetName = document.getElementById('cadet-name-input')?.value || "Cadet Hero";
        const docRef = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'settings', 'profile');
        
        // Ensure totalStars exists and is loaded safely on window scope
        const stars = typeof window.totalStars !== 'undefined' ? window.totalStars : 0;

        await setDoc(docRef, {
          stationsCompleted: window.stationsCompleted || [false, false, false, false],
          totalStars: stars,
          cadetName: cadetName,
          lastUpdated: new Date().toISOString()
        }, { merge: true });
      } catch (e) {
        console.error("Failed backing up progress", e);
      }
    }

    // Expose helper globally
    window.saveProgressToCloud = saveProgressToCloud;

    // Trigger on boot
    initFirebaseStorage();
  </script>

    let voiceActive = true;

    if ('SpeechRecognition' in window || 'webkitSpeechRecognition' in window) {

      recognition = new SpeechRec();

    function toggleVoice() {
      voiceActive = !voiceActive;
      const icon = document.getElementById('voice-icon');
      const btn = document.getElementById('voice-btn');
      if (voiceActive) {
        icon.className = "fa-solid fa-volume-high";
        btn.className = "w-10 h-10 md:w-12 md:h-12 rounded-full bg-blue-500 hover:bg-blue-600 text-white flex items-center justify-center text-base md:text-xl shadow-lg border-2 border-white shrink-0";
        speakNow("Voice is on!");
      } else {
        icon.className = "fa-solid fa-volume-xmark";
        btn.className = "w-10 h-10 md:w-12 md:h-12 rounded-full bg-slate-400 hover:bg-slate-500 text-white flex items-center justify-center text-base md:text-xl shadow-lg border-2 border-white shrink-0";
        if ('speechSynthesis' in window) {
          window.speechSynthesis.cancel();
        }
      }
    }

    function speakNow(text) {
      if (!voiceActive) return;
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel(); 
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 0.75; // Slower rate for Year 1 pupils
        utterance.pitch = 1.25; // Friendly higher pitch
        utterance.volume = 1.0; // Maximum output volume
        window.speechSynthesis.speak(utterance);
      }
    }

    function playSynthBeep(correct) {
      const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.connect(gain);
      gain.connect(audioCtx.destination);

      if (correct) {
        osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); // C5
        osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.12); // E5
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime); // Louder sound effect
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);
      } else {
        osc.type = "sawtooth";
        osc.frequency.setValueAtTime(180, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.3);
        gain.gain.setValueAtTime(0.35, audioCtx.currentTime); // Louder error chime
        osc.start();
        osc.stop(audioCtx.currentTime + 0.35);
      }
    }

    function renderOptionCards(question) {
      const container = document.getElementById('game-options-container');
      container.innerHTML = "";

      question.options.forEach((optText, index) => {
        const card = document.createElement('button');
        card.className = "bg-white border-4 border-slate-200 rounded-3xl p-6 flex flex-col items-center justify-center gap-4 transition-all hover:scale-[1.03] hover:border-indigo-400 shadow-md focus:outline-none min-h-[180px]";
        
        card.onclick = () => selectOption(index);

        let optionGraphics = "";
        let textLabel = optText;
        
        if (question.charLeft && index === 0) {
          optionGraphics = `<div class="w-28 h-28">${CharacterSVGs[question.charLeft]()}</div>`;
        } else if (question.charRight && index === 1) {
          optionGraphics = `<div class="w-28 h-28">${CharacterSVGs[question.charRight]()}</div>`;
        } else if (question.isNumber) {
          let starIcons = "";
          const digit = parseInt(optText.replace(/[^0-9]/g, '')) || (optText.includes("one") ? 1 : optText.includes("two") ? 2 : optText.includes("three") ? 3 : optText.includes("four") ? 4 : optText.includes("five") ? 5 : optText.includes("six") ? 6 : optText.includes("seven") ? 7 : optText.includes("eight") ? 8 : optText.includes("nine") ? 9 : 10);
          for(let s=0; s<digit; s++) {
            starIcons += `<i class="fa-solid fa-star text-yellow-400 text-2xl md:text-3xl animate-pulse drop-shadow"></i>`;
          }
          optionGraphics = `<div class="flex flex-wrap gap-2 justify-center max-w-[200px] py-2">${starIcons}</div>`;
          textLabel = ""; 
        } else {
          optionGraphics = `<div class="w-18 h-18 bg-blue-100 rounded-full flex items-center justify-center text-blue-500 text-3xl"><i class="fa-solid fa-question"></i></div>`;
        }

        card.innerHTML = `
          ${optionGraphics}
          ${textLabel ? `<span class="font-kids text-2xl text-slate-800 uppercase tracking-wide mt-1">${textLabel}</span>` : ''}
        `;
        container.appendChild(card);
      });
    }

    // SPEAKING STATION: SPEECH TO TEXT ENGINE
    function startSpeechCapture() {
      if (isListening) return;

      if (!recognition) {
        skipVoiceVerify();
        return;
      }

      try {
        recognition.start();
      } catch (e) {
        skipVoiceVerify();
      }
    }

    function evaluateSpeechInput(spokenText) {
      const question = trainingProgram[currentStationIdx].activities[currentActivityIdx].questions[currentQuestionIdx];
      const targetClean = question.targetText.toLowerCase().replace(/[^a-z0-9 ]/g, '').trim();

      const feedbackBox = document.getElementById('round-feedback');
      const headline = document.getElementById('feedback-headline');
      const expl = document.getElementById('feedback-explanation');

      // Flexible Year 1 evaluation: match target or essential sentence keywords
      let isMatch = spokenText.includes(targetClean) || targetClean.includes(spokenText);
      
      if (!isMatch && question.keywords) {
        let matchedCount = 0;
        question.keywords.forEach(kw => {
          if (spokenText.includes(kw.toLowerCase())) matchedCount++;
        });
        if (matchedCount >= Math.min(2, question.keywords.length)) {
          isMatch = true;
        }
      }

      if (isMatch) {
        playSynthBeep(true);
        feedbackBox.className = "py-5 px-6 rounded-3xl font-kids text-xl flex flex-col gap-2 items-center justify-center border-4 bg-emerald-50 border-emerald-400 text-emerald-800 shadow-lg";
        headline.innerText = "Awesome! 📣";
        expl.innerText = `You said: "${question.targetText}"!`;
        currentFeedbackText = `Awesome! You said: "${question.targetText}"!`;

        stationStarsEarned++;
        window.totalStars++;
        updateHUDAwards();
        if (window.saveProgressToCloud) window.saveProgressToCloud();
      } else {
        playSynthBeep(false);
        feedbackBox.className = "py-5 px-6 rounded-3xl font-kids text-xl flex flex-col gap-2 items-center justify-center border-4 bg-red-50 border-red-400 text-red-800 shadow-lg";
        headline.innerText = "Try Again! 🚒";
        expl.innerText = question.hintWrong;
        currentFeedbackText = "Try Again! " + question.hintWrong;
      }

      feedbackBox.classList.remove('hidden');
      speakNow(currentFeedbackText);
    }
